\section{$\G2$-structures on the open orbit}
The $\G2$-structure $(\g2,\psi)$ on $(0,\pi/3)\times M$ is given by
\begin{align}
  \label{eq:phifromsu3}  \g2&=\Omega_++dt\wedge \omega,\\
 \label{eq:psifromsu3}   \psi=*\g2&= \frac{1}{2}\omega^2-dt\wedge \Omega_-.
\end{align}
From above using $d(\omega^2)=0$ we get that 
\begin{align*}
    d\g2&=d\Omega_++dt\wedge(\dot{\Omega}_+ +  d\omega),\\
    d\psi&= dt\wedge ( \dot\omega\wedge\omega-d\Omega_-).
\end{align*}




\noindent
\textbf{Conditions for co-closed $\G2$-structure}
For $d\psi=0$ we have
\begin{align*}
    q_2-q_7=2\dot f f ,\quad q_3-q_8=2\dot f f ,\quad q_5-q_4=2\dot f f. 
\end{align*}
{\red{No conditions on other functions.}}

There are infinitely many solutions. For any $f>0$ we can choose
\begin{align*}
    q_2&=q_3=q_5=-q_4=-q_7=-q_8=\dot f f.
\end{align*} Then the condition $\Omega_+\wedge\Omega_-=k\omega^3$ gives
\begin{align*}
    (p_1q_6-p_6q_1)+\dot f f (p_2+p_3+p_4+p_5+p_7+p_8)=6kf^3.
\end{align*}If we set $p_1q_6-p_6q_1=0$ then $p_2=p_3=p_4=p_5=p_7=p_8=\frac{kf^2}{\dot f}$ is a solution.

\medskip

\noindent
\textbf{Conditions for nearly parallel $\G2$-structure}

Now we want to find solutions for functions $f,p_i,q_i$ such that \[d\g2=\lambda\psi\] for some non-zero real $\lambda$. The conditions on the functions are given by
\begin{align*}
&\dot p_1=\lambda q_1, \quad \dot p_6=\lambda q_6 , \quad \dot p_i=f(t)+ \lambda q_i, \quad i=2,3,4,5,7,8, \\
    &p_2 - p_7 = \lambda f(t)^2, \quad p_3 - p_8 = \lambda f(t)^2, \quad p_5 - p_4 = \lambda f(t)^2.
    \end{align*}
    
One of the possible solutions for any $\lambda\neq 0$ (with $2\Omega_+\wedge\Omega_-+\omega^3=0$) is given by
\begin{align*}
f(t)&=\sqrt{\frac{2}{\lambda}}t, \quad p_i=t^2,  i=1,2,3,5,6, \quad p_i=-t^2,  i=4,7,8,\\
q_i&=\frac{2}{\lambda}t, i=1,2, \quad q_i=\frac{1}{\lambda}\left(2-\sqrt{\frac{2}{\lambda}}\right)t,  i=2,3,5, \quad  q_i=-\frac{1}{\lambda}\left(2+\sqrt{\frac{2}{\lambda}}\right)t, i=4,7,8.
    \end{align*}
{\red{Since $t\in (0,\pi/3)$, $f>0$.}}    

    For the above solution 
    \begin{align*}
        \g2=&\sqrt{\frac{2}{\lambda}}t\ dt\wedge  (e^{12}+e^{34}+e^{56}) + t^2(e^{135}+e^{136}+e^{145}+e^{235}+e^{246}-e^{146}-e^{236}-e^{245}),\\
        \psi=& \frac{2t^2}{\lambda}(e^{1234}+e^{1256}+e^{3456})+\frac{2t}{\lambda}dt\wedge(e^{135}+e^{246}) \\ &+\frac{1}{\lambda}\left(2-\sqrt{\frac{2}{\lambda}}\right)t \ dt\wedge (e^{136}+e^{145}+e^{235}) + \frac{1}{\lambda}\left(2+\sqrt{\frac{2}{\lambda}}\right)t \ dt\wedge (e^{146}+e^{236}+e^{245}).
    \end{align*}
  
  \subsection{General solution}
  
  Any $\Z_2^2$-invariant 3-form on the principal open set is given by 
  
  \begin{equation}\label{eq:gen-phi}
  \begin{aligned}
         \g2=&(p_1e^{12}+p_2e^{34}+p_3e^{56})\wedge dt \\ &+ q_1e^{135}+q_2e^{136}+q_3e^{145}+q_4e^{146}+q_5e^{235}+q_6e^{236}+q_7e^{245}+q_8e^{246}
         \end{aligned}
     \end{equation} 
where the coefficients $p_i,q_i$ are fucntions of $t\in (0,\pi/3)$.
Using the $3$-form above one can compute the symmetric bi-linear tensor 
\[(B_{\g2})_{ij}\vol=-\frac{1}{6}(e_i\lrcorner\g2)\wedge(e_j\lrcorner\g2)\wedge\g2.\]
The $3$-form $\phi$ defines a $\G2$-structure if and only if $B_\g2$ is positive definite. In case it is positive definite we get the metric $g_\g2$ induced from $\g2$ by 
\[g_\g2=\frac{B_\g2}{\det{B_\g2}^\frac{1}{9}}.\]

For notational convenience let 
\begin{align}\label{eqn:alpha_beta}
    \alpha_i&\coloneqq q_iq_{9-i}, i=1\dots 4,\\
    \beta_1&\coloneqq q_1q_4q_6q_7,\\
    \beta_2&\coloneqq q_2q_3q_5q_8.
\end{align}

Then the symmetric bi-linear form 
$B_\g2={\rm{diag}}(B_1,B_2,B_3,-p_1p_2p_3)$ where 
\begin{align*}
B_1&=\left(\begin{array}{c c}
    p_1(q_1q_4-q_2q_3) &\cfrac{p_1(\alpha_1+\alpha_4-\alpha_2-\alpha_3)}{2}\\
    \cfrac{p_1(\alpha_1+\alpha_4-\alpha_2-\alpha_3)}{2}& p_1(q_5q_8-q_6q_7)
\end{array}\right),\\ B_2&=\left(\begin{array}{c c}
    p_2(q_1q_6-q_2q_5) &-\cfrac{p_2(\alpha_2+\alpha_4-\alpha_1-\alpha_3)}{2}\\
     -\cfrac{p_2(\alpha_2+\alpha_4-\alpha_1-\alpha_3)}{2}&  p_2(q_3q_8-q_4q_7)
\end{array}\right), \\ B_3&=\left(\begin{array}{c c}
    p_3(q_1q_7-q_3q_5) &-\cfrac{p_3(\alpha_3+\alpha_4-\alpha_1-\alpha_2)}{2}\\
     -\cfrac{p_3(\alpha_3+\alpha_4-\alpha_1-\alpha_2)}{2}&  p_3(q_2q_8-q_4q_6)
\end{array}\right).
    \end{align*}
  if we denote by 
  \[\alpha:=\sum_{i,j=1,i<j}^4 (\alpha_i-\alpha_j)^2 - 2\sum_{i=1}^4\alpha_i^2+4(\beta_1+\beta_2)\]
  then the determinant of $B_\g2$ is given by 
    \[\det(B_\g2)=-\left(\frac{p_1p_2p_3\alpha}{4}\right)^3.\]
    The normalisation condition on the nearly half-flat $\SU(3)$-structure on $\SO(4)/\Z_2^2$ implies
    that \begin{align}\label{eqn:normalization}
        \alpha&=-4(p_1p_2p_3)^2
    \end{align}
     so we get that 
    $\det(B_\g2)=(p_1p_2p_3)^9$ and thus 
    \begin{align*}
        \det(g_\g2)&= \det(B_\g2)^{1/9}=p_1p_2p_3.
    \end{align*}
Assuming that $g_\g2$ defines a Riemannian metric one can compute the Hodge star operator $*_\g2$. The $4$-form $\psi=*_\g2\g2$ is then given by 
\begin{equation}
\begin{aligned}
    \label{eq:gen-psi}\psi=&p_1p_2 e^{1234}+p_1p_3e^{1256}+p_2p_3e^{3456}\\
    &-\frac{\alpha_1(2\alpha_1-\sum_{i=1}^4\alpha_i)+2\beta_2}{2p_1p_2p_3q_8}e^{1357}+\frac{\alpha_1(2\alpha_1-\sum_{i=1}^4\alpha_i)+2\beta_1}{2p_1p_2p_3 q_1}e^{2467} \\
    &-\frac{\alpha_2(2\alpha_2-\sum_{i=1}^4\alpha_i)+2\beta_2}{2p_1p_2p_3 q_2}e^{2457}+\frac{\alpha_2(2\alpha_2-\sum_{i=1}^4\alpha_i)+2\beta_1}{2p_1p_2p_3 q_7}e^{1367}\\
    &-\frac{\alpha_3(2\alpha_3-\sum_{i=1}^4\alpha_i)+2\beta_2}{2p_1p_2p_3 q_3}e^{2367}+\frac{\alpha_3(2\alpha_3-\sum_{i=1}^4\alpha_i)+2\beta_1}{2p_1p_2p_3 q_6}e^{1457}\\
    &-\frac{\alpha_4(2\alpha_4-\sum_{i=1}^4\alpha_i)+2\beta_2}{2p_1p_2p_3q_5}e^{1467}+\frac{\alpha_4(2\alpha_4-\sum_{i=1}^4\alpha_i)+2\beta_1}{2p_1p_2p_3 q_4}e^{2357}.
\end{aligned}
\end{equation}
The derivatives are given by
\begin{align*}
    d\g2=&(q_7-q_2)e^{1234}+(q_6-q_3)e^{1256}+(q_4-q_5)e^{3456}-q_1'e^{1357}-q_8'e^{2467}-(p_3+q_2')e^{1367}\\
    &-(p_3+q_7')e^{2457}-(p_2+q_3')e^{1457}-(p_2+q_6')e^{2367}-(p_1+q_5')e^{2357}-(p_1+q_4')e^{1467},
\end{align*} and
\begin{align*}
    d\psi=&\left\{\left(\frac{\alpha p_1p_2p_3 }{4}\right)^{1/3}\left(-\frac{p_3'}{p_3^2}+\frac{2(q_2-q_7)\left(2\alpha_2-\sum_{i=1}^4\alpha_i\right)}{\alpha}+\frac{2(\beta_1-\beta_2)}{\alpha_2}\right)+\frac{(\alpha p_1p_2p_3)'}{3p_3(2\alpha p_1p_2p_3)^{2/3}}\right\}e^{12347}\\
    &+\left\{\left(\frac{\alpha p_1p_2p_3 }{4}\right)^{1/3}\left(-\frac{p_2'}{p_2^2}+\frac{2(q_3-q_6)\left(2\alpha_3-\sum_{i=1}^4\alpha_i\right)}{\alpha}+\frac{2(\beta_1-\beta_2)}{\alpha_3}\right)+\frac{(\alpha p_1p_2p_3)'}{3p_2(2\alpha p_1p_2p_3)^{2/3}}\right\}e^{12567}\\
    &+\left\{\left(\frac{\alpha p_1p_2p_3 }{4}\right)^{1/3}\left(-\frac{p_1'}{p_1^2}+\frac{2(q_5-q_4)\left(2\alpha_4-\sum_{i=1}^4\alpha_i\right)}{\alpha}+\frac{2(\beta_1-\beta_2)}{\alpha_4}\right)+\frac{(\alpha p_1p_2p_3)'}{3p_1(2\alpha p_1p_2p_3)^{2/3}}\right\}e^{34567}.
\end{align*}
\subsubsection{System of ODEs for torsion free $\G2$ structure}
The $\G2$ structure $(\g2,\psi)$ is torsion free if it satisfies $d\g2=0$ and $d\psi=0$. Assuming $d\g2=0$ we get
\begin{subequations}\label{eqns:gen_tfg2}
\begin{equation}
p_1=-q_4',\ \ p_2=q_3', \ \ p_3=q_2'
\end{equation}
\begin{equation}
q_1'=0, \ \ q_8'=0,\ \ q_5=-q_4,\ \ q_6=-q_3,\  \ q_7=-q_2.
\end{equation}
\end{subequations}

\subsubsection{System of ODEs for nearly parallel $\G2$ structure}
If the $\G2$ structure $(\g2,\psi)$ satisfies $d\g2=\lambda\psi$ for some non-zero scalar $\lambda$ then we have 
\begin{subequations}\label{eqns:gen_ng2}
\begin{equation}
q_4+q_5= -\lambda p_2p_3,
\end{equation}
\begin{equation}
     q_3+q_6= -\lambda p_1p_3,
\end{equation}
\begin{equation}
    q_2+q_7=- \lambda p_1p_2,
\end{equation}
The ODEs for $q_i$ are given by 
\begin{equation}
    q_1'=\frac{\lambda}{2p_1p_2p_3q_8} (\alpha_1(2\alpha_1-\sum_{i=1}^4\alpha_i)+2\beta_2)
\end{equation}
\begin{equation}
    q_8'=\frac{-\lambda}{2p_1p_2p_3q_1} (\alpha_1(2\alpha_1-\sum_{i=1}^4\alpha_i)+2\beta_1)
\end{equation}
\begin{equation}
    q_2'=p_3-\frac{\lambda}{2p_1p_2p_3q_7} (\alpha_2(2\alpha_2-\sum_{i=1}^4\alpha_i)+2\beta_1)
\end{equation}
\begin{equation}
    q_7'=-p_3+\frac{\lambda}{2p_1p_2p_3q_2} (\alpha_2(2\alpha_2-\sum_{i=1}^4\alpha_i)+2\beta_2)
\end{equation}
\begin{equation}
    q_3'=p_2-\frac{\lambda}{2p_1p_2p_3q_6} (\alpha_3(2\alpha_3-\sum_{i=1}^4\alpha_i)+2\beta_1)
\end{equation}
\begin{equation}
    q_6'=-p_2+\frac{\lambda}{2p_1p_2p_3q_3} (\alpha_3(2\alpha_3-\sum_{i=1}^4\alpha_i)+2\beta_2)
\end{equation}
\begin{equation}
    q_4'=-p_1+\frac{\lambda}{2p_1p_2p_3q_5} (\alpha_4(2\alpha_4-\sum_{i=1}^4\alpha_i)+2\beta_2)
\end{equation}
\begin{equation}
    q_5'=p_1-\frac{\lambda}{2p_1p_2p_3q_4} (\alpha_4(2\alpha_4-\sum_{i=1}^4\alpha_i)+2\beta_1).
\end{equation}
\end{subequations}

where $\alpha_i,\beta_i$ are defined in \eqref{eqn:alpha_beta}.

\section{Smooth extension over the singular orbits}
We now wish to see which of the $H$-invariant $3$-forms on the open orbit $\SO{(4)}/\Z_2^2\times (0,\pi/3)$ extend to the singular orbit $Q=G/K$. Let $q=\id K\in Q$ and $V=T_qB/T_qQ$ be the normal space at $q$. The group $K$ acts by cohomogeneity-1 on $V$, that is, it acts transitively on the $S^1\subset V$. A neighbourhood of $q$ in $B$ can be identified with the total space of the normal vector bundle over $Q$ denoted by $E:=\SO(4)\times_{K}V$. Once we fix a $K$-invariant inner product on $V$, for any non-zero $v\in V$ the orbit $G\cdot v$ is a principal orbit and can be denoted by a sphere bundle $G\times_{K} S^1_{\|v\|}=G/H$ over $Q$ where $\Z_2^2=H:=Stab(v)\subset K\subset G$.   Let $h\in\Gamma(\Lambda^3(TE))$ be a $K$-invariant $3$-form on the manifold $E$. Since $E=G.V$ the section $h$ can be completely determined by its restriction on $V$, thus we can see $h$ as a $K$-invariant element in $\Gamma(\Lambda^3(TE|_V))$. The choice of an $\rm{Ad}K$-invariant reductive complement $\n$ of $\mathfrak k$ in $\fg$ induces the canonical connection on the principal bundle $G\to G/K=Q$ which in turn induces a connection on the bundle $\pi\colon E\to Q$. Thus we can decompose $TE$ into the horizontal and vertical components isomorphic to $\pi^*E$ and $\pi^*TQ$ respectively. The bundle $TE|_V$ is trivial and given by $V\times (V\oplus\n)$ hence we can think of $h$ as an $K$-equivariant smooth map from $V\to \Lambda^3(V\oplus\n)$. Using the isomorphism $V\cong\R^+\times S^1$ we can describe map $h$ in polar coordinates $(t,v)\in\R^+\times S^1$ as a 1-parameter family of smooth $K$-equivariant maps $h_t\colon S^1\to \Lambda^3(V\oplus\n)$. If we fix a $u_0\in S^1$ and denote by $W$ the space of smooth equivariant maps from $S^1$ to $\Lambda^3(V\oplus\n)$ we can define the evaluation map $\epsilon$ by
    \begin{align*}
        \epsilon\colon W&\to \Lambda^3(V\oplus\n)^{\Z_2^2}\\
        w &\mapsto w(u_0).
    \end{align*} Note that since $K$ acts transitively on $S^1$ the choice of $u_0$ is irrelevant and $\epsilon$ defines a linear isomorphism. Let $W_p\subset W$ be the space of homogeneous polynomials of degree $p$ in $W$. For any non-zero $v\in V$ multiplication by the $K$-invariant polynomial  $v\to \|v\|$ does not change the restriction of $h$ to $S^1$ and hence we can increase the degree of any $h\in W_p$ by $2$. Since the space of polynomials is dense in the space of smooth functions and the finite dimensionality of $W$ ($\dim W=\dim \Lambda^3(V\oplus\n)^{\Z_2^2}$) then we can find a basis of $\Lambda^3(V\oplus\n)^{\Z_2^2}$ such that every element in the basis corresponds to an element in $W_p$ for some $p\geq 0$. Consider the smooth curve $t\mapsto h_t$ from $\R^+\to \Lambda^3(V\oplus\n)^{\Z_2^2}\cong W$ which has a Taylor expansion at zero $h_t\sim \sum_p h_pt^p$. Eschenburg--Wang \cite[Lemma 1.1]{Eschenburg-Wang} show that the map 
    \begin{align*}
        h\colon V\setminus \{0\}\to \Lambda^3(V\oplus\n)
    \end{align*} extends to $0$ if and only if $h_p\in \epsilon( W_p)$ for all $p\geq 0$.

\begin{thm}\label{thm:IVPsoln}
    Consider the initial value problem for $y\in\R^k$ satisfying
    \begin{equation}\label{eq:IVP}
\begin{aligned}
    \dot{y}(t)&=\frac{1}{t}M_{-1}(y)+M(t,y), \\ y(0)&=y_0,
\end{aligned}
    \end{equation}
    where $M_{-1}\colon \R^k\to\R^k$ is a smooth function in the neighbourhood of $y_0$ and $M:\R\times\R^k\to\R^k$ is a smooth function in the neighbourhood of $(0,y_0)$. Then there exists a unique solution of \eqref{eq:IVP} if 
    \begin{enumerate}
        \item[i)] $M_{-1}(y_0)=0$,
        \item[ii)] $n\textup{Id}-{\rm d}_{y_0}M_{-1}$ is invertible for all $n\in\mathbb N$.
    \end{enumerate}
     Furthermore the solution depends continuously on $y_0$.  
\end{thm}

Thus in order to find a solution for the system of ODEs one need to find an initial point $y_0\in\R^k$ for which $M_{-1}(y_0)=0$.

\subsection{Singular orbit at $t=\pi/3$}

The isotropy group $K$ of the singular orbit at $t=\pi/3$ stabilizes $\frac{E_1+\sqrt{3}E_2}{2}=\frac{{\rm{diag}} (2,-1,-1)}{\sqrt{6}}$. The embedding of $K_-$ in $\rm{SO}(3)$ is given by 
\begin{align*}
   \left[ \begin{array}{ccc}
   a&0&0\\
    0&a\cos\theta & - a\sin\theta\\
   0& \sin\theta& \cos\theta
   \end{array}\right]
\end{align*} for $\theta\in [0,\pi]$ and $a\in\Z_2$. Thus $K\cong\rm{S(O(2)O(1))}\cong S^1\times\mathbb Z_2$. On $\R^4=\langle \frac{E_1+\sqrt{3}E_2}{2}\rangle^\perp=\rm{Span}\{\frac{\sqrt{3}E_1-E_2}{2},E_3,E_4,E_5\}$ the action of $K_-$ fixes $\frac{E_1+\sqrt{3}E_2}{2}$, rotates the planes spanned by $E_3, E_4$ and $\frac{\sqrt{3}E_1=E_2}{2}, E_5$ by $\theta$ and $2\theta$ respectively.

The conjugation of $K_-$ with $\gamma(\pi/3)=e^{\frac{\pi}{3}E_{12}}$ gives a rotation that fixes $E_1$, rotates by $\theta$ the plane spanned by $E_3,E_4$ and rotates by $-2\theta$ the plane spanned by $E_2,E_5$ and thus under this conjugation
 \begin{align*}
  K_-&= \left[\begin{array}{cccc}
         \cos 2\theta&0&0&\sin 2\theta\\
         0&\cos\theta&-\sin\theta&0\\
         0&a\sin\theta&a\cos\theta&0\\
         - a\sin 2\theta&0&0&a\cos2\theta
    \end{array}\right].
\end{align*} 
The singular orbit at $t=\pi/3$ is isomorphic to the one at $t=0$ which we denote by $Q$ as well. The normal space is similarly denoted by $V$ and the space $T_qQ\cong \mathfrak{n}$ is the the orthogonal reductive complement of $\mathfrak{k_-}:=\rm{Lie}(K_-)$ in $\mathfrak{so}(4)\cong\mathfrak{su}(2)\oplus\mathfrak{su}(2)$. \begin{align*}
    u_1&:=E_{25}+2E_{34}, \quad u_2:=E_{23}+E_{45} ,\quad\\
    u_3&:=E_{23}-E_{45} ,\quad u_4:=E_{24}-E_{35} ,\quad
    u_5:=E_{24}+E_{35} .
\end{align*}
The $S^1$ in $K_-$ rotates $E_{12}$ and $E_{15}$ which we denote by $v_1,v_2$ respectively. Thus $V={\rm{Span}}(v_1,v_2)$. The action of the singular group $K_-$ on the 7-dimensional space $V\oplus\n=\rm{Span}\{v_1,v_2,u_1,u_2,u_3,u_4,u_5\}$ is given by
\begin{align*}
   v_1&\mapsto \cos2\theta v_1-a\sin2\theta v_2, \\
   v_2&\mapsto  \sin2\theta v_1+a\cos2\theta v_2,\\
   u_1&\mapsto a u_1,\\
   u_2&\mapsto \cos \theta u_2-a\sin\theta u_4,\\
    u_3&\mapsto \cos 3\theta u_3+a\sin3\theta u_5,\\
     u_4&\mapsto -\sin \theta u_2+a\cos \theta u_4,\\
      u_5&\mapsto -\sin 3\theta u_3+a\cos 3\theta u_5.
\end{align*}
The space $\Lambda^{3}(V\oplus \n)^{H}$ of $H$-invariant $3$-forms on $V\oplus \n$ is an $11$-dimensional space spanned by

\[v^{12}u^1, v^1u^{23},v^1u^{45},v^2u^{24},v^2u^{25},v^2u^{34},v^2u^{35},u^{124},u^{134},u^{125},u^{135}.\]

Similar to the above discussion we can compute the polynomial $h$ corresponding to any $\alpha \in \Lambda^3(V\oplus \n)^{\Z_2^2}$ by assuming $h(1)=\alpha$.

\begin{itemize}
    \item $K_-$ acts trivially on $v^{12}u^1, v^1(u^{23}-u^{45})-v^2(u^{25}-u^{34}),u^{124},u^{135}$ hence they correspond to constant polynomials. 
    \item $(e^{i\theta},1)\in K_-$ acts by $e^{2i\theta}$ on 
    $v^2u^{24},v^2u^{35},u^{125}-u^{134},v^1(u^{23}+u^{45})-v^2(u^{25}+u^{34})$
     and by $e^{6i\theta}$ on $v^1(u^{23}+u^{45})+v^2(u^{25}+u^{34})$ hence these forms correspond to degree-1 polynomials. 
    
    \item $(e^{i\theta},1)\in K_-$ acts by $e^{4i\theta}$ on $u^{125}+u^{134},v^1(u^{23}-u^{45})+v^2(u^{25}-u^{34})$ hence these forms correspond to degree-2 polynomials. 
\end{itemize}
After the reparameterization $t=t-\pi/3$, the change of coordinates from $dt,e^1,\dots e^6$ to $v_1,v_2,u_1,\dots, u_5$ is given as follows:
\begin{align*}
    v^1&=dt, \quad v^2=\frac{t}{2}(e^5+e^6),\\
    u^1&=\frac{3e^5-e^6}{2}, \quad
  u^2=e^1, \quad u^3=e^2, \quad
   u^4=e^3, \quad u^5=e^4.
\end{align*}
In these new coordinates the $\G2$-structure $\g2$ is given by
\begin{align*}
    \g2=& \frac{p_3}{t}v^{12}u^1+p_1v^1u^{23}+p_2v^1u^{45}+\frac{-3q_2+q_1}{2t}v^2u^{24}+\frac{-3q_4+q_3}{2t}v^2u^{25}+\frac{-3q_6+q_5}{2t}v^2u^{34}+\frac{-3q_8+q_7}{2t}v^2u^{35}\\
    &+\frac{q_1+q_2}{2}u^{124}+\frac{q_3+q_4}{2}u^{125}+\frac{q_5+q_6}{2}u^{134}+\frac{q_7+q_8}{2}u^{135}.
\end{align*}
The above $\G2$-structure to extend smoothly over the singular orbit the coefficients of $v^{12}u^1,v^1(u^{23}-u^{45})-v^2(u^{25}-u^{34}),u^{124},u^{135},u^{125}+u^{134},v^1(u^{23}-u^{45})+v^2(u^{25}-u^{34})$ must be \textit{even} and coefficients of $v^2u^{24},v^2u^{35},u^{125}-u^{134},v^1(u^{23}+u^{45})-v^2(u^{25}+u^{34}),v^1(u^{23}+u^{45})+v^2(u^{25}+u^{34})$ must be \textit{odd}. Thus we get the following condition on $p_i,q_i$s':
\begin{enumerate}[i.]
    \item $p_3$ is odd
    \item $p_1-p_2$ is even, $p_1+p_2$ is odd,
    \item $q_1,q_2, q_7,q_8$ are even,
    \item $q_3+q_5$ is even, $q_3-q_5$ is odd,
    \item $q_4+q_6$ is even, $q_4-q_6$ is odd.
 \end{enumerate}
 
 Imposing the above parity relations and the well-definedness of $\g2$ at $t=0$ we can express the coefficients $q_i,i=1\dots 8$ as the following Taylor series expansions around $t=0$ for some real constants $a,b,c$ and smooth functions $y_i(t),i=1\dots 8$,

\section{Taylor series expansion around the singular orbit at $t=0$}\label{section:taylor_t=0}



In the equations \eqref{eqns:yprime} we denote the coefficients of $1/t$ in $y_i'$ by $M_{-1,i}$ and we want to solve for $y_i(0)=b_i$ so all $M_{-1,i}'s$ vanish at $t=0$. In trying to compute the $b_i's$ by hand we come across 6 equations in total for $b_1,b_2,b_4,b_5,b_6$, and $b_8$ since $b_3=b_2$ and $b_7=b_6$. For notational convenience we set 
\begin{align*}
    \frac{\lambda^{5/2}ac}{\sqrt{Q}(0)} = \frac{\mu_1\lambda^{5/2}ac}{\sqrt{3a-c}}\frac{1}{(b_2+b_6)} = \frac{\mu_1K}{b_2+b_6},
\end{align*}  where $\mu_1\in\Z_2$ and $\sqrt{(b_2+b_6)^2}=\mu_1(b_2+b_6)$.
First, we solve for $b_2,b_6$. We have
\begin{align*}
   M_{-1,2}(0)&= b_2-\sqrt{\frac{3a-c}{\lambda}} + \frac{\mu_1K}{b_2+b_6}(3b_2+b_6),\\
    M_{-1,6}(0)&= b_6+\sqrt{\frac{3a-c}{\lambda}}-\frac{3\mu_1K}{b_2+b_6}(3b_2+b_6).
\end{align*}

Setting  $3M_{-1,2}(0)+M_{-1,6}(0)=0$ we get
\begin{align*}
   3b_2+ b_6=2\sqrt{\frac{3a-c}{\lambda}}.
\end{align*}
Substituting the above in $M_{-1,2}(0), M_{-1,6}(0)$ and setting $M_{-1,2}(0)+M_{-1,6}(0)=0$ we get
\[b_2+b_6-\frac{4\mu_1K}{b_2+b_6}\sqrt{\frac{3a-c}{\lambda}}=b_2+b_6-\frac{4\mu_1\lambda^2 a c}{b_2+b_6}=0
\]

Assuming $ac<0$, we get for $\mu_1=-1$, as $\lambda>0$. The above equation implies $b_2+b_6=-2\mu \lambda \sqrt{-ac}$ for $\mu\in\Z_2$. 
\begin{align*}
    b_2&=b_3=\mu \lambda \sqrt{-ac}+\sqrt{\frac{3a-c}{\lambda}},\\
    b_6&=b_7=-3\lambda\mu\sqrt{-ac}-\sqrt{\frac{3a-c}{\lambda}}
\end{align*}
The possible ways for $Q(0)\to 0$ is if $3a-c\to 0$ or $b_2\to -b_6$. In both cases, for the equations to still be well defined we need $ac \to 0$. In the former case this means both $a,c\to 0$. If $b_2\to -b_6$ then for everything to still make sense we still need $ac\to0$. But this time this does not necessarily imply both a and c to vanish and only one of them can vanish. If $a=0$ then $c<0$, if $c=0$ then $a>0$. 

The volume of the 5-dim singular orbit $V_5$ can be computed by using the basis $u_i,i=1..5$ from \eqref{basis_t0}. We compute that \begin{align*}
    V_5&= \sqrt{-\cfrac{25 \lambda^5 (ac)^3(3b_2+b_6)^4}{(3a-c)(b_2+b_6)^4}}.
\end{align*} So one of the ways that the volume of the singular orbit can vanish is when $3b_2+b_6=0$. That forces $b_2+b_6=0$ so we get both $b_2,b_6\to 0$. But then $M_{-1,2}$ and $M_{-1.6}$ can both never vanish simultaneously unless 
\[\lim_{b_2,b_6\to 0} \frac{3b_2+b_6}{b_2-b_6}=0\] and $3a-c=0$. 

The equations for $b_i,i=1,4,5,8$ are given by (assuming $ac<0$ and $\mu_1=-1$)
\begin{align*}
    M_{-1,1}(0)&= 2b_1+\frac{K}{2c(b_2+b_6)}(a(3b_4+b_8)-c(3b_1+b_5)-2b_2(3b_2+b_6)), \\
     M_{-1,8}(0)&= 2b_8-\frac{K}{2a(b_2+b_6)}(-3a(3b_4+b_8)+3c(3b_1+b_5)+2b_6(3b_2+b_6)),\\
      M_{-1,4}(0)&= 2b_4-\frac{b_2+b_6}{\sqrt{\lambda(3a-c)}}-\frac{K}{2a(b_2+b_6)}(a(3b_4+b_8)-c(3b_1+b_5)+2b_2(3b_2+b_6)),\\
      M_{-1,5}(0)&= 2b_5+\frac{b_2+b_6}{\sqrt{\lambda(3a-c)}}-\frac{K}{2c(b_2+b_6)}(3a(3b_4+b_8)-3c(3b_1+b_5)+2b_6(3b_2+b_6)).
     \end{align*}

Setting $3M_{-1,1}+M_{-1,5}=0$ and $3M_{-1,4}+M_{-1,8}=0$ we get
\begin{align*}
    2(3b_1+b_5)+\frac{b_2+b_6}{\sqrt{\lambda(3a-c)}}-\frac{K (3b_2+b_6)^2}{c (b_2+b_6)}=0,\\
    2(3b_4+b_8)-3\frac{b_2+b_6}{\sqrt{\lambda(3a-c)}}-\frac{K (3b_2+b_6)^2}{a (b_2+b_6)}=0.
\end{align*}
So we can write $3b_1+b_5, 3b_4+b_8$ in terms of $b_2+b_6$ and $3b_2+b_6$ which we both know in terms of $a,c$. Substituting these values in $M_{-1,i}=0,i=1,4,5,8$ give $b_i,i=1,4,5,8$.
\begin{align*}
    b_1&=\frac{\mu\lambda^{3/2}a}{8\sqrt{-ac(3a-c)}}\left(\frac{\mu\lambda ac(3a+c)}{\sqrt{-\lambda a c(3a-c)}}-4\mu\sqrt{-\lambda ac(3a-c)}-4\frac{3a-c}{\lambda} \right)\\
    &=-\frac{\lambda^2a}{2}\left(\frac{(3a+c)}{4(3a-c)}+1\right)+\frac{\mu}{2c}\sqrt{-\lambda ac(3a-c)},\\
     b_4&=\frac{\lambda^2c}{2}\left(\frac{(3a+c)}{4(3a-c)}-1\right)+\frac{\mu}{2a}\sqrt{-\lambda ac(3a-c)}-\mu\sqrt{\frac{-\lambda ac}{3a-c}},\\
    b_5&=\frac{3\lambda^2a}{2}\left(\frac{(3a+c)}{4(3a-c)}+1\right)-\frac{\mu}{2c}\sqrt{-\lambda ac(3a-c)}+\mu\sqrt{\frac{-\lambda ac}{3a-c}},\\
      b_8&=-\frac{3\lambda^2c}{2}\left(\frac{(3a+c)}{4(3a-c)}-1\right)-\frac{\mu}{2a}\sqrt{-\lambda ac(3a-c)}.
\end{align*}

The two known solutions that are complete at $t=0$ are Berger and the round $S^7$ for which $\mu=-1, \lambda=6/\sqrt{5}$, and  the parameters $a=\sqrt{5}/20, c=-3\sqrt{5}/100$, and $a=\sqrt{5}/25, c=-3\sqrt{5}/5$ respectively.

\large{\textbf{First order coefficients:}}

\medskip
\noindent
We now compute $y_i'(0)$ for $i=1\dots 8$. Again we can use the fact that $y_3'(0)=-y_2'(0)$ and $y_7'(0)=-y_6'(0)$. Moreover, since $y_i$ is even for $i=1,4,5,8$, $y_i'(0)=0$ in these cases.  So overall, we just need to compute $y_2'(0)$ and $y_6'(0)$. 

From \eqref{eqns:yprime} for computing $y_2'(0),y_6'(0)$ it is enough to compute the first order derivative of $M_{-1,2},M_{-1,6}$ respectively. The equations we get are:
\begin{align*}
    2y_2'(0)&=-\frac{\lambda^{3/2}}{2}\sqrt{\frac{-ac}{3a-c}}(3y_2'(0)+y_6'(0))-\frac{1}{2\lambda^{3/2}}\sqrt{\frac{3a-c}{-ac}} (y_2'(0)-y_6'(0)),\\
     2y_6'(0)&=\frac{3\lambda^{3/2}}{2}\sqrt{\frac{-ac}{3a-c}}(3y_2'(0)+y_6'(0))-\frac{1}{2\lambda^{3/2}}\sqrt{\frac{3a-c}{-ac}} (y_2'(0)-y_6'(0)).
\end{align*}
Both the above equations are same as can be seen by computing $2(y_2'(0)-y_6'(0))$ and $2(3y_2'(0)+y_6'(0))$. We get the following relation
\begin{align}\label{eqn:y_2y_6prime}

\end{rem}

\large{\textbf{Third order coefficients:}}

Similar to the first order derivatives $y_i'''(0)=0$ for $i=1,4,5,8$ and $y_3'''(0)=-y_2'''(0)$ and $y_7'''(0)=-y_6'''(0)$ so we only need to compute $y_2'''(0)$, and $y_6'''(0)$.

Although \ref{thm:IVPsoln} does not guarantee the uniqueness of the solutions with the given initial we can still show that we get a unique solution if we also fix $y_2'(0)$ (or equivalently $y_6'(0)$) for the above solutions. We can show that all the higher order derivatives of $y_i's$ at $t=0$ can be computed in terms of the lower order derivatives, essentially $y_2'(0)$. We prove the same now.

\end{align*}

\begin{align*}

\end{equation*}

From above
\begin{align*}

\section{Maximal volume orbit}
Any 1-parameter family $(\omega(t),\gamma(t))$ of SU(3)-structures on $N^6$ with initial data $(\omega(0),\gamma(0))=(\omega_0,\gamma_0)$ satisfying the evolution equation 
\begin{align*}
    \gamma'(t)=d\omega(t)-\lambda J\gamma(t)
\end{align*} for non-zero constant $\lambda$ such that $(\omega_0,\gamma_0)$ is a nearly half-flat $\SU(3)$-structure
\begin{align*}
    d\gamma_0=\frac{\lambda}{2}\omega_0^2
\end{align*} defines a nearly parallel $\G2$-structure on $I\times N^6$ for an interval $I\subset \R$. The volume $V(t)$ of the hypersurface $\{t\}\times N$ is given by 
\begin{align*}
    V(t)&=\frac{\omega^3(t)}{6}.
\end{align*} 
In order to compute the hypersurface with maximum volume amongst the family of nearly half-flat $\SU(3)$-structures we need to compute $V'(t)$. the volume of the 
\begin{align*}
    V'(t)&=2\omega'(t)\wedge\omega^2(t).
\end{align*}
But since $(d\gamma)'(t)=d(\gamma'(t))$ the nearly half-flat condition tell us that 
\[d\gamma'(t)=\lambda\omega'(t)\wedge\omega(t)\] thus 
\begin{align*}
    V'(t)&=\frac{2}{\lambda}(d\gamma'(t))\wedge\omega(t)
\end{align*} which using the evolution equation implies 
\begin{align*}
    V'(t)&=-2d(J\gamma(t))\wedge\omega(t) = 
\frac{4}{3}w_1(t)\omega^3(t) =8w_1(t)V(t).
\end{align*} Thus the nearly half-flat $\SU(3)$-structure $(\omega(t),\gamma(t))$ is a critical point of the volume functional if and only if the intrinsic torsion form $w_1(t)$ vanishes.

Let $\bar{g}=dt^2+g_t$ be the nearly G2-metric on $I\times N^6$, $\bar\del$ be the Levi-Civita connection corresponding to $\bar{g}$ and $\nu$ be the unit normal of the family of hypersurfaces $N_t:=\{t\}\times N$. The the Weingarten operator $L(t)$ is defined for all $X\in TN$ by 
\begin{align*}
    L_t(X)&= \bar{\del}_X\nu. 
\end{align*}
The mean curvature of $N_t$ is given by $l_t=\tr{L_t}$.

Since nearly parallel $\G2$-manifolds are positive Einstein with Einstein constant $\frac{3\lambda^2}{8}$ one can use the Riccati equations and Einstein conditions to show 
\begin{align*}
    l_t'&= -\tr(L_t^2)-\frac{3}{8}\lambda^2.
\end{align*}
In terms of the metric matrix the Weingarten operator $L_t=\frac{1}{2}g^{-1}_t g'_t$. Thus $l=\frac{1}{2}\tr(g^{-1}g')=\frac{(\sqrt{\det g})'}{\sqrt{\det g}}=\frac{V'}{V}$. Now if $t_0$ is a critical point of $V(t)$ i.e. $V'(t_0)=0$ (equivalently $l_{t_0}=0$) then 
\begin{align}\label{eq:V_critical}
    V''(t_0)&=l'(t_0)V(t_0)=-(\tr(L_{t_0}^2)+\frac{3\lambda^2}{8})V(t_0)<0.
\end{align}

Thus every critical point of the volume functional is a maxima. More details are given in \cite[Section 5]{Haskins-Lorenzo}. The discussion follows verbatim in our case. 

\begin{prop}
    Let $\Psi$ be a solution of \eqref{eqns:gen_ng2} defined on $(T_1,T_2)$. Let $V$ be the orbital volume functional and $l$ be the mean curvature as defined above. Then we have for $\Lambda=\frac{3\lambda^2}{8}$
    \begin{enumerate}
    \item $|T_2-T_1|\leq \pi$.
        \item for a fix $t_0\in (0,\pi)$ such that $l(0)=\Lambda \cot(t_0)$ we have that 
        \begin{align*}
            l(t)&\leq \Lambda \cot(t+t_0) \ \ \ for \ t>0, \\
            l(t)&\geq \Lambda \cot(t+t_0) \ \ \ for \ t<0. 
        \end{align*}
        \item for the same $t_0$ as above and any $t\in(T_1,T_2)$,
        \begin{align*}
            V(t)\leq V(0)\frac{\sin^\Lambda (t+t_0)}{\sin^\Lambda(t_0)}.
        \end{align*}
    \end{enumerate}
\end{prop}
\begin{proof}
    The proof follows verbatim from \cite[Proposition 5.5]{Haskins-Lorenzo} by replacing $n=6, \Lambda=3\lambda^2/8$ ({\red{We can also set $\Lambda=5$ by choosing $\lambda=4$ although it is not necessary for the proof}}). The proof essentially uses the fact that 
\begin{align*}
    \tr(L_t^2)= \frac{1}{6}l_t^2+\tr(\hat{L}_t^2)
\end{align*} and thus $l_t$ satisfies the equation 
\begin{align}
    l_t^2+\frac{1}{6}l_t^2+\Lambda = -\tr(\hat{L}_t^2). \leq 0
\end{align}
Then one can compare $l_t$ with a solution of the Riccati equation $l_t'+\frac{1}{6}l_t^2+\Lambda=0$ found in \cite[Theorem 4.1]{Eschenburg-comparison}.
\end{proof}
